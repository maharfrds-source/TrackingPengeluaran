Pengaturan Supabase Cloud Sync

Hubungkan database gratis Anda agar sinkron di HP & Laptop

Panduan 1 Menit:
  1. Buka supabase.com, buat akun & project baru (gratis).
  2. Masuk ke menu SQL Editor di Supabase, lalu jalankan script tabel di bawah.
  3. Salin Project URL dan Anon Key dari menu Project Settings > API, tempel di atas. 
-- Jalankan di SQL Editor Supabase:
CREATE TABLE IF NOT EXISTS pengeluaranku_transactions (
  id TEXT PRIMARY KEY,
  date TEXT NOT NULL,
  category TEXT NOT NULL,
  name TEXT NOT NULL,
  amount NUMERIC NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS pengeluaranku_settings (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Buka izin akses anonim (untuk website pribadi)
ALTER TABLE pengeluaranku_transactions ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public All" ON pengeluaranku_transactions FOR ALL USING (true) WITH CHECK (true);

ALTER TABLE pengeluaranku_settings ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public All Settings" ON pengeluaranku_settings FOR ALL USING (true) WITH CHECK (true);
